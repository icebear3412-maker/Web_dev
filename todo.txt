      import { Router } from 'express';

      import { authenticate } from '~/helper/Authenticate';

      import AuthController from '~/controller/auth.controller';
      import signinValidation from '~/validation/signinValidation';
      import signupValidation from '~/validation/signupValidation';

      const AuthRouter = Router();

      AuthRouter.post(
        '/front_authenticate',
        authenticate,
        AuthController.frontUserAuthenticate
      );
      AuthRouter.post(
        '/end_authenticate',
        authenticate,
        AuthController.endUserAuthenticate
      );

      AuthRouter.post('/signup', signupValidation, AuthController.userSignUp);
      AuthRouter.post('/signin', signinValidation, AuthController.userSignIn);
      AuthRouter.post('/signout', AuthController.userSignOut);

      AuthRouter.post('/refresh_token', AuthController.generateAccessToken);

      export default AuthRouter;
